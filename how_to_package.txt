# MAKE MacOS PACKAGE ------------------

(from tutorial: https://github.com/kivy/kivy-sdk-packager/tree/master/osx)

# May be needed:
brew install python@3.10

git clone https://github.com/HaigBishop/pillar-centroid-tracker.git

git clone https://github.com/kivy/kivy-sdk-packager.git

cd kivy-sdk-packager/osx

# May be needed:
export PATH=$(echo $PATH | sed 's|$HOME/miniconda/bin:||g')

pip install -r requirements.txt

# (find link for .dmg at https://github.com/kivy/kivy/releases)

curl -O -L https://github.com/kivy/kivy/releases/download/2.3.0/Kivy.dmg
hdiutil attach Kivy.dmg -mountroot .

cp -R Kivy/Kivy.app PCT.app

rm -rf PCT.app/Contents/Resources/venv
python3.10 -m venv PCT.app/Contents/Resources/venv
source PCT.app/Contents/Resources/venv/bin/activate

# May be needed:
pip install --upgrade pip setuptools wheel

# May be needed:
pip install delocate distlib

./fix-bundle-metadata.sh PCT.app -n PCT -v "1.1.1" -a "Haig Bishop" -i "/Users/haig/Documents/pillar-centroid-tracker/resources/icon.icns"

cp -R ../../pillar-centroid-tracker PCT.app/Contents/Resources/

cp ../../pillar-centroid-tracker/requirements.txt ./my-app-requirements.txt

python prepare-wheels.py --requirements-file my-app-requirements.txt --output-folder my-app-wheels

pushd PCT.app/Contents/Resources/venv/bin
source activate
popd

# May be needed:
pip install --upgrade certifi
export SSL_CERT_FILE=$(python -m certifi)

# May be needed:
brew install pkg-config openblas
export LDFLAGS="-L/opt/homebrew/opt/openblas/lib"
export CPPFLAGS="-I/opt/homebrew/opt/openblas/include"
export PKG_CONFIG_PATH="/opt/homebrew/opt/openblas/lib/pkgconfig"

cat > PCT.app/Contents/Resources/script <<EOF
#!/usr/bin/env bash
DIR="\$(cd "\$(dirname "\${BASH_SOURCE[0]}")" && pwd)"
"\$DIR/venv/bin/python" "\$DIR/pillar-centroid-tracker/pillar_centroid_tracker.py"
EOF
chmod +x PCT.app/Contents/Resources/script

SITE_PACKAGES_DIR=$(python -c "import site; print(site.getsitepackages()[0])")
pip install --platform macosx_11_0_universal2 --find-links=./my-app-wheels --no-deps --target $SITE_PACKAGES_DIR -r my-app-requirements.txt

# May be needed:
rm -rf PCT.app/Contents/Resources/pillar-centroid-tracker/.git
xattr -cr PCT.app

./cleanup-app.sh PCT.app

# Ensure we have certificates (https://developer.apple.com/account/resources/certificates/)

codesign --sign "Developer ID Application: Haig Bishop (6LR2726467)" --deep --force --verbose PCT.app

codesign --verify --deep --strict --verbose=2 PCT.app

pushd PCT.app/Contents/Resources/
ln -sf ./venv/bin/python yourapp
popd

./relocate.sh PCT.app

zip -r PCT.zip PCT.app
xcrun altool --notarize-app --primary-bundle-id "science.haig.pct" --username "your-apple-id" --password "app-specific-password" --file PCT.zip
xcrun stapler staple PCT.app

./create-osx-dmg.sh PCT.app PCT
